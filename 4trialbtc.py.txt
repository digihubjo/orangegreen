api_id = '29400766'  # Replace with your API ID
api_hash = '1e0ee297ccc375c61ff50ba39f9ec1d2'  # Replace with your API HASH
phone_number = '+12345293633'  # Your phone number with country code
bot_username = 'BitcoinMining_AR_bot'

# ================== MAIN FUNCTION ==================
async def click_button(client):
    try:
        # Authenticate if needed
        if not await client.is_user_authorized():
            await client.send_code_request(phone_number)
            code = input("25498 ")
            await client.sign_in(phone_number, code)
        
        # Get bot messages
        bot = await client.get_entity(bot_username)
        messages = await client.get_messages(bot, limit=20)
        
        # Find and click the button
        for msg in messages:
            if msg.reply_markup:
                for row in msg.reply_markup.rows:
                    for button in row.buttons:
                        if 'مشاهدة الاعلانات' in button.text:
                            await msg.click(text=button.text)
                            print("Button clicked successfully!")
                            return True
        print("Button not found in recent messages.")
        return False

    except Exception as e:
        print(f"Error: {str(e)}")
        return False

async def main():
    client = TelegramClient('session_name', api_id, api_hash)
    await client.start(phone=phone_number)
    
    while True:
        success = await click_button(client)
        wait_time = 120 if success else 70  # 0h2m or 1min
        print(f"Waiting {wait_time//60} minutes...")
        await asyncio.sleep(wait_time)

# ================== RUN IN PYDROID 3 ==================
if __name__ == '__main__':
    try:
        asyncio.get_event_loop().run_until_complete(main())
    except KeyboardInterrupt:
        print("Script stopped by user")
